import { isPublishedCharacter } from '../lib/publication';
import { getCollection } from 'astro:content';
import { CADETS, CADET_COUNT, PUBLISHER } from '../data/empyrean';
const characters = await getCollection('characters', isPublishedCharacter);
const profiles = new Map(characters.map(entry => [entry.data.fullName, entry]));
const withProfiles = CADETS.flatMap(cadet => {
  const profile = profiles.get(cadet.name);
  return profile ? [{ cadet, profile }] : [];
});
const withoutProfiles = CADETS.filter(cadet => !profiles.has(cadet.name));
const retrievedOn = '2026-10-25';
